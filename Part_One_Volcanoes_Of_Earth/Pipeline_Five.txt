Title: All volcanoes with the highest and lowest elevation by volcano types

[{
 $project: {
  _id: 0,
  Country: 1,
  Volcano_Name: 1,
  Volcano_Type: 1,
  elevation_meters: 1
 }
}, {
 $group: {
  _id: '$Volcano_Type',
  Highest_Elevation: {
   $max: '$elevation_meters'
  },
  Lowest_Elevation: {
   $min: '$elevation_meters'
  }
 }
}, {
 $out: 'HighestAndLowestElevationByVolcanoTypes'
}]
